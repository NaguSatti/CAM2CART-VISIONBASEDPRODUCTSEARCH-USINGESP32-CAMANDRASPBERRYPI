import os
import cv2
import numpy as np
import pytesseract
import requests
from PIL import Image
import pymongo
from urllib.parse import quote_plus

ESP_IP = "10.203.140.253"

JPG_URL = f"http://{ESP_IP}/jpg"
STREAM_URL = f"http://{ESP_IP}/stream"

OUT = os.path.expanduser("~/project/out")
os.makedirs(OUT, exist_ok=True)

SNAP = os.path.join(OUT, "snap.jpg")
QR = os.path.join(OUT, "qr.png")


def get_jpg():
    try:
        r = requests.get(JPG_URL, timeout=5)
        arr = np.frombuffer(r.content, np.uint8)
        return cv2.imdecode(arr, cv2.IMREAD_COLOR)
    except:
        return None


def get_stream():
    try:
        r = requests.get(STREAM_URL, stream=True, timeout=6)
        buf = b""

        for chunk in r.iter_content(2048):
            buf += chunk

            a = buf.find(b'\xff\xd8')
            b = buf.find(b'\xff\xd9', a + 2)

            if a != -1 and b != -1:
                jpg = buf[a:b + 2]
                arr = np.frombuffer(jpg, np.uint8)
                return cv2.imdecode(arr, cv2.IMREAD_COLOR)

    except:
        return None

    return None


def ocr(img):
    g = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    pil = Image.fromarray(g)

    text = pytesseract.image_to_string(
        pil,
        config="--oem 3 --psm 6"
    )

    text = "".join(
        c if c.isalnum() or c == " " else " "
        for c in text
    )

    return " ".join(text.split())


def make_qr(url):
    import qrcode

    img = qrcode.make(url)
    img.save(QR)


def main():
    print("Trying /jpg ...")

    img = get_jpg()

    if img is None:
        print("Trying /stream ...")
        img = get_stream()

    if img is None:
        print("FAILED to get image")
        return

    cv2.imwrite(SNAP, img)

    print("Saved:", SNAP)

    text = ocr(img)

    print("Text:", text)

    pymongo.save(text)

    if not text:
        print("No text detected")
        return

    url = "cam2cart.vercel.app"

    make_qr(url)

    print("QR saved:", QR)
    print("URL:", url)


if __name__ == "__main__":
    main()
